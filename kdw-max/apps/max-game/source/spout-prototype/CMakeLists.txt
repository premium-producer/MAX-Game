cmake_minimum_required(VERSION 3.20)
project(kdw_spout_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 required")
endif()
if(NOT DEFINED SPOUT_ROOT OR NOT DEFINED NODE_HEADERS OR NOT DEFINED NODE_LIB)
  message(FATAL_ERROR "Pass pinned SPOUT_ROOT, NODE_HEADERS, NODE_LIB")
endif()
set(GL "${SPOUT_ROOT}/SPOUTSDK/SpoutGL")
set(DX "${SPOUT_ROOT}/SPOUTSDK/SpoutDirectX/SpoutDX")
# Probe-local frame counting, without changing the user's global Spout registry.
# The SDK otherwise disables its frame semaphore when SpoutSettings was never run.
file(READ "${GL}/SpoutFrameCount.cpp" FRAME_SOURCE)
set(FRAME_MARKER "\t// If frame counting is disabled, set the new frame flag true")
string(FIND "${FRAME_SOURCE}" "${FRAME_MARKER}" MARKER_POSITION)
if(MARKER_POSITION LESS 0)
  message(FATAL_ERROR "Pinned Spout frame-counter patch no longer applies")
endif()
string(REPLACE "${FRAME_MARKER}" "\tm_bFrameCount = true; // KDW probe: process-local only, never write registry\n${FRAME_MARKER}" FRAME_SOURCE "${FRAME_SOURCE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/SpoutFrameCount-local.cpp" "${FRAME_SOURCE}")
add_library(spout STATIC "${DX}/SpoutDX.cpp" "${GL}/SpoutCopy.cpp" "${GL}/SpoutDirectX.cpp" "${CMAKE_CURRENT_BINARY_DIR}/SpoutFrameCount-local.cpp" "${GL}/SpoutSenderNames.cpp" "${GL}/SpoutSharedMemory.cpp" "${GL}/SpoutUtils.cpp")
target_compile_definitions(spout PUBLIC SPOUT_BUILD_STATIC NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_include_directories(spout PUBLIC "${DX}" "${GL}")
target_link_libraries(spout PUBLIC d3d11 dxgi d3d9 d3dcompiler shlwapi winmm version psapi)
add_library(kdw_spout MODULE bridge.cpp)
target_include_directories(kdw_spout PRIVATE "${NODE_HEADERS}")
target_compile_definitions(kdw_spout PRIVATE NAPI_VERSION=8 NODE_GYP_MODULE_NAME=kdw_spout)
target_link_libraries(kdw_spout PRIVATE spout "${NODE_LIB}" delayimp)
target_link_options(kdw_spout PRIVATE /DELAYLOAD:node.exe)
set_target_properties(kdw_spout PROPERTIES PREFIX "" SUFFIX ".node")
add_executable(kdw_spout_receiver receiver.cpp)
target_link_libraries(kdw_spout_receiver PRIVATE spout)
